\noindent\colorbox{azulClaro}{\parbox{\dimexpr\textwidth-2\fboxsep}{%
 {\fontsize{42}{50}\selectfont\color{uniAzul}\bfseries Pregunta}\par
\vspace{0.2cm}
{\fontsize{38}{46}\selectfont ¿Qué cambia al exigir evidencia de que una vacante remota admite personas ubicadas en Colombia?}
}}

\vspace{1cm}
\noindent
\begin{minipage}[t][68cm][s]{0.485\textwidth}
\vspace{0pt}
{\fontsize{52}{62}\selectfont\color{uniAzul}\bfseries Problema}\par
{\fontsize{40}{49}\selectfont El filtro original del flujo de n8n acepta ``Remote'' o ``Remoto'' como señal de ubicación válida. La modalidad de trabajo no indica qué países admite el empleador. Una vacante remota restringida a otro país puede terminar en una alerta dirigida a Colombia.}

\vfill
{\fontsize{52}{62}\selectfont\color{uniAzul}\bfseries Prototipo}\par
{\fontsize{40}{49}\selectfont Búsqueda de ofertas $\longrightarrow$ normalización $\longrightarrow$ clasificación geográfica $\longrightarrow$ máximo cinco elegibles $\longrightarrow$ correo HTML.}
\par\vspace{0.25cm}
{\fontsize{36}{44}\selectfont Cada anuncio recibe una de tres salidas. Solo una ubicación confirmada para Colombia llega al correo; la incertidumbre queda visible para revisión.}

\vspace{0.3cm}
\begin{center}
\fcolorbox{uniAzul}{azulClaro}{\parbox{0.86\linewidth}{\centering
{\fontsize{36}{43}\selectfont\bfseries\color{uniAzul}COLOMBIA}\par
{\fontsize{31}{38}\selectfont Evidencia explícita o rótulo mundial}\par\vspace{0.2cm}
{\fontsize{36}{43}\selectfont\bfseries\color{uniRojo}NO COLOMBIA}\par
{\fontsize{31}{38}\selectfont Restricción a otros países}\par\vspace{0.2cm}
{\fontsize{36}{43}\selectfont\bfseries REVISAR}\par
{\fontsize{31}{38}\selectfont Ubicación ambigua}
}}
\end{center}

\vfill
{\fontsize{52}{62}\selectfont\color{uniAzul}\bfseries Datos y método}\par
{\fontsize{40}{49}\selectfont Benchmark dirigido de 16 anuncios públicos consultados el 24/11/2026 en Remotavia. Ocho admiten Colombia (``Colombia only'' u ``Open worldwide'') y ocho están restringidos a otros países o regiones. Se comparan la regla ``Remote/Remoto'' del flujo original y una regla que exige elegibilidad explícita.}
\par\vspace{0.25cm}
{\fontsize{36}{44}\selectfont La referencia de clase es la etiqueta geográfica del anuncio. El conjunto quedó balanceado deliberadamente para observar falsos positivos y falsos negativos. Se calcula precisión, sensibilidad y exactitud por regla.}

\vspace{0.6cm}
{\fontsize{32}{39}\selectfont La muestra es deliberada y pequeña. Evalúa el error de clasificación en estos casos, no estima la frecuencia de ofertas elegibles en LinkedIn ni en el mercado laboral.}
\end{minipage}\hfill
\begin{minipage}[t][68cm][s]{0.485\textwidth}
\vspace{0pt}
{\fontsize{52}{62}\selectfont\color{uniAzul}\bfseries Resultados}\par
{\fontsize{40}{49}\selectfont Las 16 vacantes dicen ``Remote''. La regla original acepta todas y produce ocho falsos positivos. La regla mejorada acepta las ocho que indican Colombia o apertura mundial y rechaza las ocho restringidas.}
\par\vspace{0.3cm}
{\fontsize{36}{44}\selectfont Ejemplo observado: ``Remote in set countries -- United States only'' contiene la palabra ``Remote'', pero el anuncio no admite residencia en Colombia.}

{\fontsize{52}{62}\selectfont\color{uniAzul}\bfseries Conclusión}\par
{\fontsize{40}{49}\selectfont En el benchmark, distinguir el lugar autorizado de la modalidad remota elimina ocho falsos positivos y mantiene sensibilidad de 100\%. El flujo operativo propone tres estados y deja para revisión los anuncios sin evidencia suficiente.}

\vspace{0.5cm}
{\fontsize{32}{39}\selectfont La evaluación usa etiquetas de un agregador, no anuncios recolectados con Apify desde LinkedIn. Para validar el flujo en LinkedIn se requiere contrastar la extracción textual con revisión manual.}
